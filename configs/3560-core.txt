! ============================================================
! Практика 04. Коммутатор 3560-24PS (Core) - агрегирование LACP
! Вводить в CLI с нуля (сразу после Enter в окне CLI)
! ============================================================
enable
configure terminal

! чтобы опечатки не вызывали паузу "Translating ..." (экономит время)
no ip domain-lookup
hostname Core

! ---- Канал 1: SW1. Порты Fa0/1-2 ----
interface range fa0/1-2
 channel-protocol lacp
 channel-group 1 mode active
 exit

! ---- Канал 2: SW2. Порты Fa0/3-4 ----
interface range fa0/3-4
 channel-protocol lacp
 channel-group 2 mode active
 exit

! ---- Канал 3: SW3. Порты Fa0/5-6 ----
interface range fa0/5-6
 channel-protocol lacp
 channel-group 3 mode active
 exit

end
write memory

! ============================================================
! ПРОВЕРКА
!   show etherchannel summary
!   show lacp neighbor
!
! Ожидается:
!   Group  Port-channel  Protocol    Ports
!   1      Po1(SU)         LACP        Fa0/1(P)  Fa0/2(P)
!   2      Po2(SU)         LACP        Fa0/3(P)  Fa0/4(P)
!   3      Po3(SU)         LACP        Fa0/5(P)  Fa0/6(P)
! ============================================================
