! ============================================================
! Практика 04. Коммутатор 2960-24TT - агрегирование LACP
! ОДИН и тот же скрипт на SW1, SW2, SW3.
! Меняется ТОЛЬКО строка hostname.
! ============================================================
enable
configure terminal

no ip domain-lookup
hostname SW1

! ---- Единственный канал этого коммутатора: Fa0/1-2 -> Core ----
interface range fa0/1-2
 channel-protocol lacp
 channel-group 1 mode passive
 exit

end
write memory

! ============================================================
! ПРОВЕРКА
!   show etherchannel summary
!
! Ожидается:
!   Group  Port-channel  Protocol    Ports
!   1      Po1(SU)         LACP        Fa0/1(P)  Fa0/2(P)
!
! ВАЖНО: именно passive, потому что на 3560 стоит active.
!        passive + passive не поднимется; active + active тоже можно.
! ============================================================
